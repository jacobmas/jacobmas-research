\[\lam(\matA) = \{\vecy \in \Z^m : \vecy = \matA^{t}\vecs \bmod{q} \text{ for some } \vecs \in \Z^n \}\]

\[\lamperp(\matA) = \{ \vece \in \Z^m : \matA\vece = \veczero \bmod {q} \}\]

\[\lamperp = q \cdot \lam^*\quad\text{ and }\quad \lam = q \cdot (\lamperp)^*\]
